```go
package repository

import (
    "context"

    "github.com/aws/aws-sdk-go-v2/aws"
    "github.com/aws/aws-sdk-go-v2/service/s3"
    "github.com/jackc/pgx/v5/pgxpool"
    "github.com/redis/go-redis/v9"
)

type healthRepo struct {
    pool   *pgxpool.Pool
    redis  *redis.Client
    s3     *s3.Client
    bucket string
}

func NewHealthRepository(pool *pgxpool.Pool, redisClient *redis.Client, s3Client *s3.Client, bucket string) HealthRepository {
    return &healthRepo{pool: pool, redis: redisClient, s3: s3Client, bucket: bucket}
}

func (r *healthRepo) PingDB(ctx context.Context) error {
    return r.pool.Ping(ctx)
}

func (r *healthRepo) PingRedis(ctx context.Context) error {
    return r.redis.Ping(ctx).Err()
}

func (r *healthRepo) PingS3(ctx context.Context) error {
    _, err := r.s3.HeadBucket(ctx, &s3.HeadBucketInput{Bucket: aws.String(r.bucket)})
    return err
}
```

**3. Create `apps/gateway/handlers/health.go`:**

```go
package handlers

import (
    "net/http"

    "github.com/gin-gonic/gin"
    "go.uber.org/zap"

    "gateway/repository"
)

type HealthHandler struct {
    healthRepo repository.HealthRepository
    logger     *zap.Logger
}

func NewHealthHandler(healthRepo repository.HealthRepository, logger *zap.Logger) *HealthHandler {
    return &HealthHandler{healthRepo: healthRepo, logger: logger}
}

type HealthStatus struct {
    Status   string            `json:"status"`
    Services map[string]string `json:"services"`
}

func (h *HealthHandler) HandleHealth(c *gin.Context) {
    ctx := c.Request.Context()
    services := make(map[string]string)
    allHealthy := true

    // Check DB
    if err := h.healthRepo.PingDB(ctx); err != nil {
        services["database"] = "unhealthy"
        allHealthy = false
        h.logger.Error("health check: database unhealthy", zap.Error(err))
    } else {
        services["database"] = "healthy"
    }

    // Check Redis
    if err := h.healthRepo.PingRedis(ctx); err != nil {
        services["redis"] = "unhealthy"
        allHealthy = false
        h.logger.Error("health check: redis unhealthy", zap.Error(err))
    } else {
        services["redis"] = "healthy"
    }

    // Check S3
    if err := h.healthRepo.PingS3(ctx); err != nil {
        services["storage"] = "unhealthy"
        allHealthy = false
        h.logger.Error("health check: storage unhealthy", zap.Error(err))
    } else {
        services["storage"] = "healthy"
    }

    status := "healthy"
    statusCode := http.StatusOK
    if !allHealthy {
        status = "degraded"
        statusCode = http.StatusServiceUnavailable
    }

    c.JSON(statusCode, HealthStatus{Status: status, Services: services})
}
```

**Verify and update the `Delete` method in `apps/gateway/services/document.go`.**

The document service's Delete method (created in Plan 01) must follow this exact sequence per user decision:

1. Get S3 key from `docRepo.GetS3Key(ctx, docID, userID)` -- if document not found, return appropriate error
2. Attempt S3 deletion: call `storage.Client.DeleteObject(ctx, &s3.DeleteObjectInput{Bucket: aws.String(storage.Bucket), Key: aws.String(s3Key)})`
3. If S3 deletion fails: log a warning using `logger.Warn("failed to delete S3 object, orphaned object may remain", zap.String("s3_key", s3Key), zap.Error(err))` -- do NOT return error, proceed to DB deletion
4. Delete from DB: call `docRepo.Delete(ctx, docID)` -- this is a single DELETE FROM documents WHERE id = $1, and ON DELETE CASCADE in the migration handles all child tables (pages, segments, translations, project_metadata)
5. If DB deletion fails: return error

The implementation should look like:
```go
func (s *documentService) Delete(ctx context.Context, docID, userID string) error {
    // Verify ownership and get S3 key
    s3Key, err := s.docRepo.GetS3Key(ctx, docID, userID)
    if err != nil {
        return fmt.Errorf("document not found: %w", err)
    }

    // Attempt S3 cleanup (best effort per user decision)
    if s3Key != "" {
        _, err = s.storage.Client.DeleteObject(ctx, &s3.DeleteObjectInput{
            Bucket: aws.String(s.storage.Bucket),
            Key:    aws.String(s3Key),
        })
        if err != nil {
            s.logger.Warn("failed to delete S3 object, orphaned object may remain",
                zap.String("s3_key", s3Key),
                zap.String("doc_id", docID),
                zap.Error(err),
            )
            // Continue with DB deletion per user decision
        }
    }

    // Delete from DB (CASCADE handles child tables)
    if err := s.docRepo.Delete(ctx, docID); err != nil {
        return fmt.Errorf("failed to delete document from database: %w", err)
    }

    s.logger.Info("document deleted",
        zap.String("doc_id", docID),
        zap.String("user_id", userID),
    )
    return nil
}
```

Key points:
- The old code in handlers/document.go had a manual cascade delete with a transaction (DELETE FROM translations, DELETE FROM source_segments, DELETE FROM pages, DELETE FROM documents). This is now replaced by a single `DELETE FROM documents WHERE id = $1` because the migration has `ON DELETE CASCADE` on all child tables.
- S3 deletion is synchronous and happens BEFORE DB deletion (per user decision).
- If S3 fails, we log and continue. An orphaned S3 object is acceptable.
- The handler (document.go) should NOT have any S3 or SQL logic -- it just calls `docService.Delete(ctx, docID, userID)`.
